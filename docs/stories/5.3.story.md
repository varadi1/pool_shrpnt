# Story 5.3: Template Manager with Version Control

## Status
Ready for Review

## Story
**As an** Admin,
**I want** to create, edit, version, and manage folder templates with diff viewing and rollback capabilities,
**so that** I can maintain consistent folder structures across orders while preserving existing deployments

## Acceptance Criteria
1. Admin can create new folder templates with hierarchical structure definition
2. Template editing creates new version without affecting existing orders
3. Version comparison shows visual diff between any two template versions
4. Rollback functionality allows reverting to previous template versions
5. Template validation ensures required fields and naming conventions
6. Template preview shows folder structure before saving
7. Templates can be marked as default, deprecated, or archived
8. Template usage statistics show which orders use which template version
9. All template operations are audited with version history

```typescript
// types/templates.ts
interface Template {
  id: string;
  name: string;
  description: string;
  status: 'draft' | 'active' | 'default' | 'deprecated' | 'archived';
  currentVersion: string;
  createdBy: string;
  createdAt: Date;
  updatedAt: Date;
  usageCount: number;
  tags: string[];
}

interface TemplateVersion {
  id: string;
  templateId: string;
  version: string; // Semantic version: 1.0.0
  structure: FolderNode;
  changelog: string;
  author: string;
  publishedAt: Date;
  isPublished: boolean;
  usageCount: number;
}

interface FolderNode {
  id: string;
  name: string;
  path: string; // e.g., "/root/documents/financial"
  type: 'folder' | 'file';
  children: FolderNode[];
  properties: FolderProperties;
  permissions: PermissionSet;
  metadata: Record<string, any>;
}

interface FolderProperties {
  namingPattern?: string; // Regex for validation
  description?: string;
  required: boolean;
  locked: boolean;
  lockSchedule?: {
    t3?: boolean;
    t1?: boolean;
    t0?: boolean;
    t8?: boolean;
  };
  maxSize?: number; // MB
  allowedFileTypes?: string[];
}

interface PermissionSet {
  inherit: boolean;
  breakInheritance: boolean;
  groups: {
    groupId: string;
    groupName: string;
    permission: 'read' | 'write' | 'full';
  }[];
}

interface DiffResult {
  summary: {
    added: number;
    removed: number;
    modified: number;
    unchanged: number;
  };
  changes: DiffNode[];
}

interface DiffNode {
  path: string;
  name: string;
  type: 'added' | 'removed' | 'modified' | 'unchanged';
  oldValue?: FolderNode;
  newValue?: FolderNode;
  propertyChanges?: PropertyChange[];
  children?: DiffNode[];
}

interface PropertyChange {
  field: string;
  oldValue: any;
  newValue: any;
  type: 'added' | 'removed' | 'modified';
}
```

**Diff Algorithm Specification**
```typescript
// services/templateDiff.ts
class TemplateDiffService {
  /**
   * Compare two template versions
   * Algorithm:
   * 1. Build path maps for both versions
   * 2. Find added paths (in new, not in old)
   * 3. Find removed paths (in old, not in new)
   * 4. Find common paths and compare properties
   * 5. Build hierarchical diff tree
   */
  calculateDiff(oldVersion: FolderNode, newVersion: FolderNode): DiffResult {
    const oldPaths = this.buildPathMap(oldVersion);
    const newPaths = this.buildPathMap(newVersion);
    
    const added = this.findAddedPaths(oldPaths, newPaths);
    const removed = this.findRemovedPaths(oldPaths, newPaths);
    const modified = this.findModifiedPaths(oldPaths, newPaths);
    
    return this.buildDiffTree(added, removed, modified);
  }
  
  private compareProperties(old: FolderProperties, new: FolderProperties): PropertyChange[] {
    const changes: PropertyChange[] = [];
    const allKeys = new Set([...Object.keys(old), ...Object.keys(new)]);
    
    for (const key of allKeys) {
      if (JSON.stringify(old[key]) !== JSON.stringify(new[key])) {
        changes.push({
          field: key,
          oldValue: old[key],
          newValue: new[key],
          type: !old[key] ? 'added' : !new[key] ? 'removed' : 'modified'
        });
      }
    }
    return changes;
  }
}
```
